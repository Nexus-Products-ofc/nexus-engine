1. Como começar

git clone https://github.com/heitorhanzelgontijo-svg/nexus-engine.git
cd nexus-engine
pip install pygame numpy
python main.py
python main.py --editor
pip install ursina && python main.py --3d
python build.py

Exemplo mínimo

from nexus_engine import Application, World, Transform, SpriteRenderer, Camera, Debug

world = World("Jogo")
cam = world.create_entity("Cam")
cam.add_component(Transform(0, 0))
cam.add_component(Camera())
player = world.create_entity("Player")
player.add_component(Transform(100, 100))
player.add_component(SpriteRenderer(color=(100, 200, 255), width=32, height=32))
Debug.log("Pronto")
Application(title="Meu Jogo", backend="pygame").run(world)

2. Funções principais (API)

Application / Cenas

from nexus_engine import Application, SceneManager, ProjectSettings, NexusEngine
settings = ProjectSettings.instance()
world = SceneManager.create_scene("Level1")
Application(title="MeuJogo", backend="pygame", show_splash=True).run(world)

ECS / Física / Input

from nexus_engine import Component, Transform, SpriteRenderer, Rigidbody2D, BoxCollider2D, Vector2, Input, Audio, Debug

class Mover(Component):
    def update(self, dt):
        self.transform.x += 40 * dt

e = world.create_entity("Player")
e.add_component(Transform(100, 100))
e.add_component(SpriteRenderer(color=(120, 180, 255), width=32, height=32))
e.add_component(Rigidbody2D(mass=1.0))
e.add_component(BoxCollider2D(size=Vector2(32, 48)))
e.add_component(Mover())
if Input.get_key_down("space"):
    Audio.play_sfx("jump.wav")
Debug.show_fps = True
